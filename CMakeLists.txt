cmake_minimum_required(VERSION 3.21)
project(DirOpen VERSION 0.10.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
qt_add_library(diropen_ui STATIC src/filepanel.cpp src/filepanel.h src/mainwindow.cpp src/mainwindow.h src/fileoperation.cpp src/fileoperation.h src/operationdialog.cpp src/operationdialog.h src/sessionstore.cpp src/sessionstore.h src/filetransfer.cpp src/filetransfer.h src/filelistview.cpp src/filelistview.h)
target_sources(diropen_ui PRIVATE src/filesearch.cpp src/filesearch.h src/searchdialog.cpp src/searchdialog.h src/favoritesdialog.cpp src/favoritesdialog.h)
target_sources(diropen_ui PRIVATE src/folderview.cpp src/folderview.h src/folderviewsdialog.cpp src/folderviewsdialog.h)
target_sources(diropen_ui PRIVATE src/previewloader.cpp src/previewloader.h src/previewwidget.cpp src/previewwidget.h)
target_sources(diropen_ui PRIVATE src/imagepreview.cpp src/imagepreview.h src/thumbnailcache.cpp src/thumbnailcache.h src/thumbnailview.cpp src/thumbnailview.h)
target_sources(diropen_ui PRIVATE src/foldercomparison.cpp src/foldercomparison.h src/comparisondialog.cpp src/comparisondialog.h src/windowspath.h)
target_link_libraries(diropen_ui PUBLIC Qt6::Widgets)
target_include_directories(diropen_ui PUBLIC src)
target_compile_definitions(diropen_ui PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(diropen_ui PUBLIC DIROPEN_VERSION="${PROJECT_VERSION}")
target_link_libraries(diropen_ui PRIVATE ole32 shell32 uuid user32 windowsapp)
qt_add_executable(DirOpen WIN32 src/main.cpp)
target_link_libraries(DirOpen PRIVATE diropen_ui)
if(MSVC)
  target_compile_options(diropen_ui PRIVATE /utf-8 /W4)
  target_compile_options(DirOpen PRIVATE /utf-8 /W4)
endif()
